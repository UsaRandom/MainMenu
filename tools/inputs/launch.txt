# Launch a game with cheats on, and watch the loading screen.
#
# Under an emulator there is no cart to stream a ROM onto, so this exercises everything up to the
# handoff -- cheat flattening, play history, the progress bar -- and stops short of boot(). See
# simulate_load() in screen_launch.c for why that is the honest behaviour rather than a test hook.
#
# Golden Eye 007 is index 5 in the N64 tab (1080, AeroFighters, Banjo, Conkers, Donkey Kong 64,
# Golden Eye) and is one of the fixture games the cheat database actually covers. A game with no
# cheats reaches the same loading screen and proves nothing about the cheat path -- the first
# version of this script picked Donkey Kong 64, which has none, and its Z press did nothing.
#
# A is deliberately not used on the detail sheet: A is Play there, not "open cheats". The first
# version used it to mean the latter and launched the game four steps early.

wait 300            # boot plate, then let the first rows decode

press right x5      # Golden Eye 007
press a             # detail sheet
wait 40
fbdump              # 1: the sheet, showing its cheat count

press z             # cheats screen
wait 30
press a             # tick the first group
press down
press a             # and a second, so the emitter has more than one to flatten
wait 20
fbdump              # 2: two groups ticked

press b             # back to the sheet
wait 40
press l             # canary launch: the one gesture that still stops on the diagnostic page,
                    # now that Start boots for real again. L latches (mode=canary, diagnose)
                    # on the sheet -- see launch_set_mode() -- so this exercises the latch AND
                    # the page in one press.
wait 12
fbdump              # 3: mid-fade -- the sheet dimming, not a cleared screen

wait 12
fbdump              # 4: further into the fade, so it must be darker than frame 3

wait 180            # let it finish -- and stop on the cheat diagnostic page
fbdump              # 5: the diagnostic, on screen_fault, which has no update function and no way
                    #    to be dismissed. Under ares there is no cart so the page says so; what is
                    #    under test here is that it DRAWS AND STAYS. Four earlier versions of this
                    #    page did neither on hardware, and none of them was ever looked at here.

wait 120            # two seconds of nothing pressed and nothing changing
fbdump              # 6: still there, identical to frame 5

exit
