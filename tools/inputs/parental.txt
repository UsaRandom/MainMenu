# Set a parental code, lock a game, then try to launch it.
#
# The whole feature end to end, because every part of it is a state machine that only means
# anything in sequence: a code that is set but guards nothing, or a lock on a menu with no code,
# both look fine in a single frame.
#
# The code is six C presses -- cup cdown cleft cright cup cdown -- because the alphabet is now the
# four C directions and nothing else. Six of four is 4,096, the same space the old four of eight
# gave. A code entered with A/Z/L/R would no longer be accepted at all.

wait 120

press start         # settings
wait 20
fbdump              # 1: the settings screen itself

# Exactly the number of rows to Parental controls, not a number large enough to clamp there.
# Clamping hid a whole change once: removing two rows moved this row from index 4 to index 2 and
# `press down x4` went on landing on it, so the suite reported byte-identical hashes for a screen
# that had lost a third of its contents.
#
# Now x5, because folding the sound screen back into Settings added three rows -- effects, music
# and track -- between Theme and Clock. This count is expected to move whenever that list does,
# and a run that suddenly dumps the wrong screen is the count being wrong, not the screen.
press down x5       # -> Parental controls
press a             # no code yet, so straight in
wait 20
fbdump              # 2: the panel, nothing set

press a             # Code -> the pad, asking for a new one
wait 20
press cup
press cdown
press cleft
press cright
press cup
press cdown
wait 20
fbdump              # 3: entered once, now asking again

press cup
press cdown
press cleft
press cright
press cup
press cdown
wait 20
fbdump              # 4: back on the panel, code set

press down          # -> Locked games
press a
wait 20
press a             # lock the first title
press down
press a             # and the second
wait 15
fbdump              # 5: the lock list, two padlocks

press b             # panel
wait 15
press b             # settings
wait 15
press b             # grid
wait 20
press r x2          # Recent -> Favourites -> N64, where the two locked titles sort first
wait 200            # the tiles have to decode before a badge on one means anything
fbdump              # 6: the grid, padlocks on the two locked tiles

press a             # the detail sheet for a locked game
wait 40
fbdump              # 7: the sheet, with its Locked row and the C-left hint

press a             # try to launch it
wait 25
fbdump              # 8: the code pad, refusing

press cup
press cdown
press cleft
press cright
press cup
press cup           # deliberately wrong on the sixth press
wait 25
fbdump              # 9: rejected, and that guess is now on the card

# Five seconds of penalty for the one wrong entry above, and the pad says so rather than simply
# ignoring presses. 60 frames is a second, so this frame lands well inside the wait.
wait 60
fbdump              # 10: the wait, counting down

# 5 s is 300 frames; overshoot so the exact frame the wait expires on is not load-bearing.
wait 340
press cup
press cdown
press cleft
press cright
press cup
press cdown        # the real code
wait 40

exit
