# Add a player, give them a face and two colours, and switch to them.
#
# The whole feature end to end, and it has to be end to end for the same reason parental.txt does:
# every part of this is a state machine that only means something in sequence. A roster with two
# names in it and an active profile that never moved looks correct in any single frame.
#
# Note what this run does NOT prove. Under ares the storage prefix is the ROM's own DFS, which is
# read only, so every profile_save() here fails and the roster exists only in RAM. That is the
# documented behaviour of the whole cache layer (see library/cache.h) and it is what makes the
# screens testable at all -- but "the roster survives a power cycle" is a claim only hardware can
# settle. Same gap as every other written file in this program.
#
# It also cannot reach the destructive half. Removing a player deletes their saves, and the
# fixture has none to delete. tools/hosttest is where the deletion itself is exercised, against
# real files.

wait 150

press l x8          # along the tab rail. L past the FIRST tab IS the picker -- no A, no chip focus
wait 30
fbdump              # 1: ten cards under a live tab rail, straight from one press past FAVORITES

press right         # -> slot 2, empty
press a             # fills it and opens the keyboard
wait 30
fbdump              # 2: QWERTY with a digit row on top, and SPACE / DONE only -- no DELETE key

press b             # nothing typed, so B leaves rather than doing nothing
wait 30
fbdump              # 3: back on the cards; the slot exists but is still unnamed

press a             # the slot is filled now, so A plays as them
wait 30
fbdump              # 4: the grid, chip showing the new unnamed player

press l x8          # back to the picker, which opens on the active profile -- slot 2 already
press z             # the edit menu: Name, then Icon and colours
press down
press a             # -> the appearance editor
wait 40
fbdump              # 5: categories, sprites, and TWO labelled colour rows under them

# The cursor is not the choice. A takes the sprite under it; moving on must not un-take it, or
# there is no way to hold a face still while trying colours against it -- which is the only thing
# this screen is for.
press right x2
press a             # take this one: accent bar under its cell, and the preview picks it up
press right x3      # and walk away from it
wait 40
fbdump              # 6: cursor three cells past the accent bar, preview still on the chosen sprite

# Six, not five. The editor opens on the page holding the profile's own icon and puts the cursor
# on it, and Player 2's starter is the fox -- Animals cell 14, which is row 1 of five. Crossing the
# grid from there is three presses and passing through the Icon row is two more, but the starter is
# baked by tools/mkiconmeta.py against whatever corpus shipped and can land on any row. Six reaches
# the Plate row from all of them; the surplus presses fall off the bottom and do nothing.
press down x6       # off the grid, through the Icon row, onto Plate
press right x3      # amber -> purple
wait 30
fbdump              # 7: the Plate row live, and the 60 px preview repainted on the new plate

press up            # -> the Icon row
# Left, not right. Player 2's plate is amber, so their artwork already defaults to black, which is
# the second-to-last swatch -- pressing right moves one square and then stops against the end, and
# a row that does not appear to move is not a row that has been tested.
press left x7       # black -> red
wait 30
fbdump              # 8: the Icon row live, the Plate row still showing its choice dimmed

press start         # accept. A is only for taking a sprite now; START and B both commit.
wait 30
fbdump              # 9: back on the cards, wearing the chosen sprite and both colours

exit
